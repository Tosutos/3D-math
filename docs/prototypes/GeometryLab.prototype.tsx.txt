"use client";

import { Canvas } from "@react-three/fiber";
import { ContactShadows, Edges, Html, OrbitControls } from "@react-three/drei";
import { useMemo, useState } from "react";

type ShapeId = "cube" | "cuboid" | "cylinder" | "cone" | "sphere";

const SHAPES: Array<{
  id: ShapeId;
  name: string;
  subtitle: string;
  faces: string;
  edges: string;
  vertices: string;
}> = [
  { id: "cube", name: "정육면체", subtitle: "모든 모서리의 길이가 같은 입체", faces: "6개", edges: "12개", vertices: "8개" },
  { id: "cuboid", name: "직육면체", subtitle: "마주 보는 면이 서로 같은 상자 모양", faces: "6개", edges: "12개", vertices: "8개" },
  { id: "cylinder", name: "원기둥", subtitle: "위아래가 같은 원인 입체", faces: "3개", edges: "2개", vertices: "0개" },
  { id: "cone", name: "원뿔", subtitle: "원 모양 밑면과 한 꼭짓점을 가진 입체", faces: "2개", edges: "1개", vertices: "1개" },
  { id: "sphere", name: "구", subtitle: "중심에서 겉면까지 거리가 모두 같은 입체", faces: "1개", edges: "0개", vertices: "0개" },
];

function VertexDots({ shape }: { shape: ShapeId }) {
  const points = useMemo(() => {
    if (shape === "cube") return [-1, 1].flatMap((x) => [-1, 1].flatMap((y) => [-1, 1].map((z) => [x, y, z] as [number, number, number])));
    if (shape === "cuboid") return [-1.5, 1.5].flatMap((x) => [-0.85, 0.85].flatMap((y) => [-0.65, 0.65].map((z) => [x, y, z] as [number, number, number])));
    if (shape === "cone") return [[0, 1.25, 0] as [number, number, number]];
    return [];
  }, [shape]);

  return (
    <>
      {points.map((point, index) => (
        <mesh key={`${point.join("-")}-${index}`} position={point}>
          <sphereGeometry args={[0.055, 16, 16]} />
          <meshStandardMaterial color="#ffcf33" emissive="#8a5b00" emissiveIntensity={0.45} />
        </mesh>
      ))}
    </>
  );
}

function ShapeMesh({ shape, showEdges, showVertices }: { shape: ShapeId; showEdges: boolean; showVertices: boolean }) {
  const material = <meshStandardMaterial color="#62d5ff" roughness={0.38} metalness={0.08} transparent opacity={0.88} />;

  return (
    <group rotation={[0.08, -0.35, 0]}>
      <mesh castShadow receiveShadow>
        {shape === "cube" && <boxGeometry args={[2, 2, 2]} />}
        {shape === "cuboid" && <boxGeometry args={[3, 1.7, 1.3]} />}
        {shape === "cylinder" && <cylinderGeometry args={[1, 1, 2.25, 64]} />}
        {shape === "cone" && <coneGeometry args={[1.15, 2.5, 64]} />}
        {shape === "sphere" && <sphereGeometry args={[1.2, 64, 32]} />}
        {material}
        {showEdges && <Edges color="#f6fbff" linewidth={2} threshold={15} />}
      </mesh>
      {showVertices && <VertexDots shape={shape} />}
    </group>
  );
}

export default function GeometryLab() {
  const [shape, setShape] = useState<ShapeId>("cube");
  const [showEdges, setShowEdges] = useState(true);
  const [showVertices, setShowVertices] = useState(true);
  const selected = SHAPES.find((item) => item.id === shape)!;

  return (
    <main className="min-h-screen overflow-hidden bg-[#07111f] text-slate-50">
      <section className="relative mx-auto flex min-h-screen w-full max-w-7xl flex-col gap-8 px-5 py-6 md:px-10 lg:grid lg:grid-cols-[390px_1fr] lg:items-center">
        <div className="pointer-events-none absolute inset-0 -z-0 bg-[radial-gradient(circle_at_20%_20%,rgba(55,211,255,0.24),transparent_34%),radial-gradient(circle_at_80%_0%,rgba(255,207,51,0.16),transparent_30%),linear-gradient(135deg,#07111f,#10182c_45%,#07111f)]" />
        <div className="pointer-events-none absolute inset-0 -z-0 opacity-[0.16] [background-image:linear-gradient(rgba(255,255,255,.18)_1px,transparent_1px),linear-gradient(90deg,rgba(255,255,255,.18)_1px,transparent_1px)] [background-size:42px_42px]" />

        <aside className="relative z-10 rounded-[2rem] border border-white/12 bg-white/10 p-5 shadow-2xl shadow-cyan-950/40 backdrop-blur-xl md:p-6">
          <p className="mb-3 inline-flex rounded-full border border-cyan-200/20 bg-cyan-200/10 px-3 py-1 text-xs font-semibold tracking-[0.24em] text-cyan-100">
            3D MATH LAB
          </p>
          <h1 className="text-4xl font-black tracking-[-0.06em] md:text-5xl">입체도형 조작실</h1>
          <p className="mt-4 text-sm leading-7 text-slate-300">
            마우스나 손가락으로 도형을 돌리고, 확대하고, 면·모서리·꼭짓점을 바로 확인하는 학습용 3D 실험실입니다.
          </p>

          <div className="mt-6 grid grid-cols-2 gap-2">
            {SHAPES.map((item) => (
              <button
                key={item.id}
                onClick={() => setShape(item.id)}
                className={`rounded-2xl border px-3 py-3 text-left text-sm font-bold transition ${
                  shape === item.id
                    ? "border-cyan-200 bg-cyan-200 text-slate-950 shadow-lg shadow-cyan-500/20"
                    : "border-white/10 bg-white/[0.06] text-slate-200 hover:border-cyan-200/60 hover:bg-white/12"
                }`}
              >
                {item.name}
              </button>
            ))}
          </div>

          <div className="mt-5 flex gap-2">
            <button
              onClick={() => setShowEdges((value) => !value)}
              className={`flex-1 rounded-2xl px-3 py-3 text-sm font-bold ${showEdges ? "bg-white text-slate-950" : "bg-white/10 text-slate-200"}`}
            >
              모서리 {showEdges ? "ON" : "OFF"}
            </button>
            <button
              onClick={() => setShowVertices((value) => !value)}
              className={`flex-1 rounded-2xl px-3 py-3 text-sm font-bold ${showVertices ? "bg-yellow-300 text-slate-950" : "bg-white/10 text-slate-200"}`}
            >
              꼭짓점 {showVertices ? "ON" : "OFF"}
            </button>
          </div>

          <div className="mt-6 rounded-3xl border border-white/10 bg-slate-950/45 p-5">
            <h2 className="text-2xl font-black tracking-[-0.04em]">{selected.name}</h2>
            <p className="mt-2 text-sm text-slate-300">{selected.subtitle}</p>
            <dl className="mt-5 grid grid-cols-3 gap-2 text-center">
              <div className="rounded-2xl bg-white/8 p-3"><dt className="text-xs text-slate-400">면</dt><dd className="mt-1 font-black">{selected.faces}</dd></div>
              <div className="rounded-2xl bg-white/8 p-3"><dt className="text-xs text-slate-400">모서리</dt><dd className="mt-1 font-black">{selected.edges}</dd></div>
              <div className="rounded-2xl bg-white/8 p-3"><dt className="text-xs text-slate-400">꼭짓점</dt><dd className="mt-1 font-black">{selected.vertices}</dd></div>
            </dl>
          </div>
        </aside>

        <section className="relative z-10 h-[58vh] min-h-[430px] overflow-hidden rounded-[2.4rem] border border-white/12 bg-slate-950/50 shadow-2xl shadow-black/40 lg:h-[78vh]">
          <Canvas shadows camera={{ position: [4, 3, 5], fov: 42 }}>
            <color attach="background" args={["#07111f"]} />
            <ambientLight intensity={0.9} />
            <directionalLight position={[4, 6, 3]} intensity={2.1} castShadow />
            <pointLight position={[-3, -1, -2]} intensity={1.1} color="#4fd8ff" />
            <ShapeMesh shape={shape} showEdges={showEdges} showVertices={showVertices} />
            <ContactShadows position={[0, -1.45, 0]} opacity={0.35} scale={7} blur={2.5} />
            <Html position={[0, 1.95, 0]} center>
              <div className="rounded-full border border-white/15 bg-black/35 px-4 py-2 text-sm font-bold text-cyan-50 shadow-xl backdrop-blur-md">
                드래그해서 회전 · 휠/핀치로 확대
              </div>
            </Html>
            <OrbitControls enablePan enableZoom minDistance={3} maxDistance={8} />
          </Canvas>
        </section>
      </section>
    </main>
  );
}
